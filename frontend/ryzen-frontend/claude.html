-- Run in Supabase SQL editor

CREATE TABLE orders (
  id             SERIAL PRIMARY KEY,
  sender_id      TEXT NOT NULL,
  name           TEXT NOT NULL,
  drink          TEXT NOT NULL,
  milk           TEXT NOT NULL,
  sweetness      TEXT NOT NULL,
  building       TEXT NOT NULL,
  status         TEXT NOT NULL DEFAULT 'pending',
  price          INTEGER NOT NULL,
  created_at     TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE TABLE config (
  id               INTEGER PRIMARY KEY DEFAULT 1,
  daily_limit      INTEGER NOT NULL DEFAULT 15,
  accepting_orders BOOLEAN NOT NULL DEFAULT true,
  cutoff_message   TEXT NOT NULL DEFAULT 'Sorry, orders are full for today! Try again tomorrow.',
  pickup_time      TEXT NOT NULL DEFAULT '8:00–8:15am, SJH corridor'
);

The dashboard API routes need protection — add a simple secret header check so only you can call them. Not OAuth, not JWT — just an ADMIN_SECRET env variable checked on every /api/* route. Good enough for this scale.
// Simple admin middleware — add this to all /api routes
const adminAuth = (req, res, next) => {
  const secret = req.headers['x-admin-secret']
  if (secret !== process.env.ADMIN_SECRET) {
    return res.status(401).json({ error: 'Unauthorized' })
  }
  next()
}

// Usage:
app.get('/api/orders/today', adminAuth, async (req, res) => {
  // only you get here
})

How order limits work in the bot

// Every time someone tries to place an order,
// check BEFORE starting the conversation:

async function isAcceptingOrders() {
  // Get config row
  const { data: config } = await supabase
    .from('config')
    .select('accepting_orders, daily_limit')
    .eq('id', 1)
    .single()

  if (!config.accepting_orders) return false

  // Count today's non-cancelled orders
  const today = new Date().toISOString().split('T')[0] // "2025-06-15"

  const { count } = await supabase
    .from('orders')
    .select('*', { count: 'exact', head: true })
    .gte('created_at', `${today}T00:00:00`)
    .neq('status', 'cancelled')

  return count < config.daily_limit
}

// In your handleMessage, IDLE state:
case STATES.IDLE:
  const open = await isAcceptingOrders()
  if (!open) {
    const { data: config } = await supabase
      .from('config').select('cutoff_message').eq('id',1).single()
    await send(senderId, config.cutoff_message)
    break
  }
  sessions[senderId] = { state: STATES.ASK_NAME, data: {} }
  await send(senderId, "Hi! What's your name?")
  break

The dashboard folder is served as static files from Express. No React, no Vite, no build step — just the HTML/CSS/JS you already know. Express serves it at /dashboard. You protect it with a password prompt in the JS that sends the admin secret header.
